cmake_minimum_required(VERSION 3.20)
project(ApexGfxSnapshotPoolExperiment LANGUAGES CXX)
enable_testing()
add_executable(GfxSnapshotPoolTests SnapshotPointerPagePoolTests.cpp)
target_compile_features(GfxSnapshotPoolTests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(GfxSnapshotPoolTests PRIVATE /W4 /WX /permissive- /EHsc)
else()
  target_compile_options(GfxSnapshotPoolTests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
find_package(Threads REQUIRED)
target_link_libraries(GfxSnapshotPoolTests PRIVATE Threads::Threads)
add_test(NAME GfxSnapshotPoolTests COMMAND GfxSnapshotPoolTests)
